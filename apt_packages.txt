# Preferred Ubuntu 22.04 package versions bootstrap.bash installs with apt-get.
#
# One pinned package specification per line. The upstream version is fixed;
# the trailing wildcard admits Ubuntu security/build revisions for that same
# version. On Ubuntu releases that do not provide these versions, the bootstrap
# installs the release's available versions instead. Comments and blank lines
# are ignored.

# Bare container images (e.g. ubuntu:22.04) ship apt-get but not these, and the
# bootstrap and its installers assume curl / python3 / git / tar are present.
curl=7.81.0-*
python3=3.10.6-*
git=1:2.34.1-*
# Pin git's version-locked documentation dependency too. This lets apt replace
# a newer Git PPA installation with the configured Ubuntu release version.
git-man=1:2.34.1-*
tar=1.34+dfsg-*
# lifeboat (the home-directory backup tool) compresses with pigz when present,
# falling back to gzip; pigz parallelises gzip across cores on big home dirs.
pigz=2.6-*
gawk=1:5.1.0-*
ripgrep=13.0.0-*
tmux=3.2a-*
# Pi uses fd for file discovery and accepts Ubuntu's fdfind binary name.
fd-find=8.3.1-*
# The autocuda agent plugin's report skills render their markdown reports to
# HTML by shelling out to pandoc (`autocuda report html`).
pandoc=2.9.2.1-*
# autocuda's `viz graph` renderer depends on pygraphviz, a C binding with no
# prebuilt wheel: it compiles against the system Graphviz headers
# (graphviz/cgraph.h, from libgraphviz-dev) and links libcgraph, using a C
# compiler and pkg-config. build-essential supplies the compiler toolchain.
graphviz=2.42.2-*
libgraphviz-dev=2.42.2-*
pkg-config=0.29.2-*
build-essential=12.9ubuntu3
# The Brev installer (install-latest.sh) invokes `sudo` unconditionally; we
# install it even when running as root, where sudo is a no-op passthrough.
sudo=1.9.9-*
# HTTPS downloads need the CA bundle.
ca-certificates=20260601~22.04.*
# ssh-keygen is required when either SSH key input is configured.
openssh-client=1:8.9p1-*
